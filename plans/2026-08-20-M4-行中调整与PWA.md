# M4 实施计划：行中 AI 局部重规划 + PWA 离线壳

> **For agentic workers:** 按任务顺序执行。已授权在本项目目录内 git add/commit。壳文件 `行程助手.html` 是 CRLF，用内容匹配 Edit。测试只追加不动旧段。**不得真实调用 DeepSeek API**（mock fetch）。

**Goal:** 两件收尾：(1) L2 行中局部重规划——旅途中用大白话说"D3 下雨换室内"，AI 生成替换方案，预览确认后写入 override 层，可一键还原；(2) PWA——挂到网址后可"添加到主屏幕"，离线也能打开。

**前置事实：**
- 壳内已有：`dsJSON(userPrompt,opts)`（全局，DeepSeek JSON 调用）、`state`（mountTrip 初始化，含 done/books/plan/me/skip）、`K(s)` 存储前缀、渲染链 renderTimeline/dayCard/renderToday、L1 的 toggleSkipDay/toggleSkipPoi 模式（override 层先例）。
- `DAYS` 是 TripPack 只读数据；dayCard(d) 渲染单天；`state.done[d.n]`/`state.skip.days[d.n]` 已存在。
- tab4（智能助手）现有卡：今日状态 `#todayBody`、天气决策、最优重排 `#optBox`、预约提醒 `#bookBox`、`.foot` 按钮区。
- DeepSeek key：`'ta-ds-key'`（load/save 全局）。

**契约（override 层）：**
- `state.ovr = load(K('ovr'),{})`：day n → 被替换的字段子集 `{title,meta,plan,know,tags}`（n/full/wd/anchor/type 永远沿用原数据，AI 不许碰）。
- `effDay(d)`：`var o=state.ovr[d.n];return o?Object.assign({},d,o,{n:d.n,full:d.full,wd:d.wd,anchor:d.anchor,type:d.type}):d;`
- renderTimeline 用 `effDay(DAYS[i])` 喂 dayCard；renderToday 命中当天也用 effDay。updateProgress 不受影响（只看 done/skip）。
- AI patch 契约（dsJSON 返回）：`{days:[{n,title,meta,plan,know,tags}],note:"一句话说明为什么这么改"}`，days 至多 3 天。

5. 函数：
```js
var adjPending=null;
function aiAdjustUi(){
 var t=(document.getElementById('adjIn').value||'').trim();
 if(!t)return;
 aiAdjust(t);
}
function aiAdjust(text){
 var box=document.getElementById('adjBox');
 if(!load('ta-ds-key','')){box.innerHTML='<div class="dim">先在雷达卡粘贴 DeepSeek Key。</div>';return Promise.resolve();}
 box.innerHTML='<div class="dim">AI 正在出替换方案(约 10-30 秒)…</div>';
 var daysCtx=DAYS.map(function(d){return {n:d.n,full:d.full,wd:d.wd,title:d.title,meta:d.meta,plan:d.plan,know:d.know};});
 var p='这是亲子自驾行程「'+META.title+'」的逐日安排 JSON:\n'+JSON.stringify(daysCtx)
  +'\n\n用户在旅途中提出调整:「'+text+'」。请只替换需要改的天数(至多 3 天),其余天不要出现在结果里。'
  +'输出 JSON:{days:[{n,title,meta,plan,know,tags}],note:"一句话说明为什么这么改"}。'
  +'要求:plan/know 是完整新内容;地点真实存在;适合 '+META.kidAge+' 岁孩子;tags 为字符串数组。只输出 JSON。';
 return dsJSON(p,{maxTokens:6000}).then(function(r){
  if(!r||!Array.isArray(r.days)||!r.days.length)throw new Error('AI 没有给出可替换的天');
  r.days=r.days.slice(0,3).filter(function(pd){return DAYS.some(function(d){return d.n===pd.n;});});
  if(!r.days.length)throw new Error('AI 返回的天数不在行程里');
  r.days.forEach(function(pd){if(Array.isArray(pd.tags))pd.tags=pd.tags.map(function(t){return typeof t==='string'?{t:t}:t;});});
  adjPending=r;
  var h='<div style="font-size:13px;line-height:1.7">';
  r.days.forEach(function(pd){
   var od=DAYS.filter(function(d){return d.n===pd.n;})[0];
   h+='<div style="border-top:1px solid var(--line);padding:8px 0"><b>D'+pd.n+'</b> <span class="dim">原:'+od.title+'</span><br>→ <b>'+pd.title+'</b><br><span class="dim">'+pd.plan.slice(0,3).join('；')+'</span></div>';
  });
  if(r.note)h+='<div class="dim" style="margin:6px 0">AI 说明:'+r.note+'</div>';
  h+='<button class="btn" onclick="applyAdjust()">应用</button> <button class="btn" onclick="discardAdjust()">放弃</button></div>';
  box.innerHTML=h;
 }).catch(function(e){box.innerHTML='<div class="dim">出方案失败:'+(e.message||e)+'</div>';throw e;});
}
function applyAdjust(){
 if(!adjPending)return;
 adjPending.days.forEach(function(pd){
  state.ovr[pd.n]={title:pd.title,meta:pd.meta,plan:pd.plan,know:pd.know,tags:pd.tags||[]};
 });
 save(K('ovr'),state.ovr);
 adjPending=null;
 document.getElementById('adjBox').innerHTML='<div class="dim">已应用,行程页对应天数带「已调整」角标,可随时还原。</div>';
 renderTimeline();renderToday();
}
function discardAdjust(){adjPending=null;document.getElementById('adjBox').innerHTML='';}
function revertDay(n){delete state.ovr[n];save(K('ovr'),state.ovr);renderTimeline();renderToday();}
```
注意：测试里直接调 `ctx.aiAdjust(...)` 和 `ctx.applyAdjust()`——`adjIn` 在 stub-DOM 里存在（getElementById 自动造），aiAdjustUi 只是取值转发。

`sw.js`：
```js
var C='ta-v1';
var SHELL=['./行程助手.html','./manifest.webmanifest','./icons/icon-192.png','./icons/icon-512.png'];
self.addEventListener('install',function(e){e.waitUntil(caches.open(C).then(function(c){return c.addAll(SHELL);}).then(function(){return self.skipWaiting();}));});
self.addEventListener('activate',function(e){e.waitUntil(self.clients.claim());});
self.addEventListener('fetch',function(e){
 if(e.request.method!=='GET')return;
 var u=new URL(e.request.url);
 if(u.origin!==self.location.origin)return;   // 跨域(瓦片/OSRM/天气/DeepSeek)不碰
 e.respondWith(caches.match(e.request).then(function(hit){
  return hit||fetch(e.request).then(function(r){
   var cp=r.clone();caches.open(C).then(function(c){c.put(e.request,cp);});
   return r;
  });
 }));
});
```
